# mapping recommendations
# dynamic mapping is not used much in prod
# set up dynamic to "strict"
# dont map strings as both text and keywords
# dont use coercion to avoid mistakes
# float is good enough for precisions than long, double
# mapping parameters - if no sorting, agg, or scripting, doc_value: false
# if no relevance: norm: false
# if no need to filter values, index: false

# 10. search basics - query DSL
# (1) simple - search everything
GET /products/_search
{
  "query":{
    "match_all":{}
  }
}

# (2) term-level search
# this ONLY applies exact matches
GET /products/_search
{
  "query": {
    "term": {
      "tags.keyword": {
        "value": "Vegetable",
        "case_insensitive": true
      }
    }
  }
}

# (3) this is a CONTAIN relationship
GET /products/_search
{
  "query": {
    "terms": {
      "tags.keyword": ["Soup", "Meat"]
    }
  }
}

# (4) range search
GET /products/_search
{
  "query": {
    "range": {
      "in_stock": {
        "gte": 1,
        "lte": 5
      }
    }
  }
}

# (5) wildcards and regex
GET /products/_search
{
  "query": {
    "regexp": {
      "tags.keyword": {
        "value": "Bee(f|r)+"
      }
    }
  }
}

# (6) whether a col exists
GET /products/_search
{
  "query": {
    "exists": {
      "field": "tags.keyword"
    }
  }
}

# (7) whether a col doesn't exist: query-bool-must_not-exist

      "must_not": [
        {
          "exists": {
            "field": "tags.keyword"

  }
}

# (12) query with a boolean

    "bool": {
      "must": [
        {
          "term": {
            "tags.keyword": "Alcohol"
          }
        }
      ]
    }
  }
}

# filter is the same as must but ignores any relevance score

  "query": {

        {
          "term": {
            "tags.keyword": "Alcohol"
          }
        }

  }
}

# (14) boosting queries
# 1.should, 2.negative,3.negative_boost pm addomg weights
GET /recipes/_search
{
  "query": {
    "boosting": {
      "positive": {
        "bool": {
          "must": [
            { "match_all": {} }
          ],
          "should": [
            {
              "term": {
                "ingredients.name.keyword": "Pasta"
              }
            }
          ]
        }
      },
      "negative": {
        "term": {
          "ingredients.name.keyword": "Bacon"
        }
      },
      "negative_boost": 0.5
    }
  }
}

# (15) disjunction max: this uses weights on matching cirtera on multi-fields

        { "match": { "tags": "vegetable" } }

      "tie_breaker": 0.3
    }
  }
}
