# Callback function to check for delivery results
def delivery_report(err, msg):
  if err is not None:
        print(f'Message delivery failed: {err}')
  else:
        print(f'Message delivered to {msg.topic()} [{msg.partition()}]')

# Function to produce messages
def produce_messages(file_path, topic_name):
    with open(file_path, 'r') as file:
        for line in file:
          producer.produce(topic_name, line.encode('utf-8'), callback=delivery_report) #use utf-8 for bianry strings
          producer.poll(1) # an async function on sending messages; 1s
